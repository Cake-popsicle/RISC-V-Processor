addi x0,  x0, 123     # no effect; x0 is always 0

addi x31, x0, 1       # pass marker

##############################/BRANCHES/##############################

addi x1, x0, 5
addi x2, x0, 5

# beq taken
beq x1, x2, beq_taken
addi x3, x0, -1       # fail, should be skipped
beq_taken:
addi x3, x0, 1        # x3 = 1 if beq taken correctly

# beq not taken
beq x1, x0, fail
addi x4, x0, 1        # x4 = 1 if beq not taken correctly

# bne taken
bne x1, x0, bne_taken
addi x5, x0, -1       # fail, should be skipped
bne_taken:
addi x5, x0, 1        # x5 = 1 if bne taken correctly

# bne not taken
bne x1, x2, fail
addi x6, x0, 1        # x6 = 1 if bne not taken correctly

# signed comparisons
addi x10, x0, -1
addi x11, x0, 1

# blt taken: -1 < 1
blt x10, x11, blt_taken
addi x7, x0, -1       # fail, should be skipped
blt_taken:
addi x7, x0, 1        # x7 = 1 if blt taken correctly

# blt not taken: 1 < -1 false
blt x11, x10, fail
addi x8, x0, 1        # x8 = 1 if blt not taken correctly

# bge taken: 1 >= -1
bge x11, x10, bge_taken
addi x9, x0, -1       # fail, should be skipped
bge_taken:
addi x9, x0, 1        # x9 = 1 if bge taken correctly

# bge not taken: -1 >= 1 false
bge x10, x11, fail
addi x12, x0, 1       # x12 = 1 if bge not taken correctly

# unsigned comparisons
# x10 = -1 = 0xFFFFFFFF unsigned
# x11 = 1

# bltu taken: 1 < 0xFFFFFFFF
bltu x11, x10, bltu_taken
addi x13, x0, -1      # fail, should be skipped
bltu_taken:
addi x13, x0, 1       # x13 = 1 if bltu taken correctly

# bltu not taken: 0xFFFFFFFF < 1 false
bltu x10, x11, fail
addi x14, x0, 1       # x14 = 1 if bltu not taken correctly

# bgeu taken: 0xFFFFFFFF >= 1
bgeu x10, x11, bgeu_taken
addi x15, x0, -1      # fail, should be skipped
bgeu_taken:
addi x15, x0, 1       # x15 = 1 if bgeu taken correctly

# bgeu not taken: 1 >= 0xFFFFFFFF false
bgeu x11, x10, fail
addi x16, x0, 1       # x16 = 1 if bgeu not taken correctly

jal x0, end

fail:
addi x31, x0, -1      # x31 = -1 means a branch failed
beq x0, x0, fail

end:
addi x31, x0, 1      # pass marker

#jal
jal x1, target
addi x2, x0, 99      # skipped
target:
addi x3, x0, 2       # target

#jalr
auipc x4, 0

# These will be the register values
# x1 = 4

# x3 = 2

# x5 = 24

# x7 = 7
